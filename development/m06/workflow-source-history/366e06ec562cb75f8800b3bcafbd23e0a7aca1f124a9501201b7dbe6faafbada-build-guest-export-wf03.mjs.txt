import { buildPresentationWf03, buildPresentationWf03Template } from './build-presentation-wf03.mjs';

export function mapGuestExportEnvelope(body, executionId) {
  const fields=['event_id','event_type','contract_version','job_id','kind','stage','input_hash','correlation_id'];
  const safe=value=>typeof value==='string'&&/^[A-Za-z0-9._:-]{1,200}$/u.test(value);
  const handlers={prepare_presentation:'native.prepare_presentation',render_presentation:'native.render_presentation',validate_presentation:'native.validate_presentation',publish_presentation:'domain.publish_presentation',render_export:'native.render_export'};
  const invalid=()=>({status:400,body:{error:{code:'DISPATCH_ENVELOPE_INVALID',message:'Use the published fixed presentation/export stage envelope.'}}});
  if(!body||typeof body!=='object'||Array.isArray(body)||Object.keys(body).length!==fields.length||fields.some(key=>!Object.hasOwn(body,key))
    ||body.event_type!=='job.stage_ready'||body.contract_version!=='1.0'||!safe(body.event_id)||!safe(body.job_id)||!safe(body.correlation_id)
    ||typeof body.input_hash!=='string'||!/^[a-f0-9]{64}$/u.test(body.input_hash)||typeof body.stage!=='string'||!Object.hasOwn(handlers,body.stage)
    ||!(body.kind==='presentation_export'&&body.stage==='render_export'||body.kind==='presentation_render'&&body.stage!=='render_export'))return invalid();
  if(!safe(executionId))return{status:500,body:{error:{code:'WF03_INTERNAL_ERROR',message:'The fixed export work order could not be prepared.'}}};
  return{status:200,body:{schema_version:'1.0',event_id:body.event_id,job_id:body.job_id,kind:body.kind,stage:body.stage,input_hash:body.input_hash,correlation_id:body.correlation_id,workflow_execution_id:executionId,handler:handlers[body.stage]}};
}
function extend(value){
  const graph=structuredClone(value);const node=graph.nodes.find(node=>node.name==='Map presentation stage');
  // Retained graph uses a deterministic EditFields node, identified by its
  // expression rather than adopting an arbitrary new node name or type.
  const mapper=node??graph.nodes.find(node=>typeof node.parameters?.jsonOutput==='string'&&node.parameters.jsonOutput.includes('mapPresentationStageEnvelope'));
  if(!mapper||mapper.type!=='n8n-nodes-base.set'||mapper.typeVersion!==3.4)throw new Error('M06_WF03_RETAINED_MAPPER_REQUIRED');
  mapper.parameters.jsonOutput=`={{ (${mapGuestExportEnvelope.toString()})($('Receive presentation stage').first().json.body, $execution.id) }}`;
  mapper.notes='Five closed stage/kind pairs: four retained presentation stages and one native guest export. Metadata only. No snapshots, source facts, bearer grants, lease tokens or result bytes enter n8n.';
  return graph;
}
export function buildGuestExportWf03(credential){return extend(buildPresentationWf03(credential));}
export function buildGuestExportWf03Template(){return extend(buildPresentationWf03Template());}
const canonical=value=>JSON.stringify(value,(_key,item)=>item&&typeof item==='object'&&!Array.isArray(item)?Object.fromEntries(Object.entries(item).sort(([a],[b])=>a.localeCompare(b))):item);
export function validateGuestExportWf03(value,credential){const expected=buildGuestExportWf03(credential);if(!value||value.name!==expected.name||['nodes','connections','settings'].some(key=>canonical(value[key])!==canonical(expected[key])))throw new Error('M06_WF03_GRAPH_UNCONFIRMED');return true;}
