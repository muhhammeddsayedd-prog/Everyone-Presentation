async function command(executable, args, { input = '', binary = false, allowFailure = false, timeout = 60000, environment } = {}) {
  return new Promise((resolve, reject) => {
    const child = spawn(executable, args, { cwd: ROOT, shell: false, windowsHide: true, env: environment, stdio: ['pipe', 'pipe', 'pipe'] });
    const out = [], err = []; let size = 0, timedOut = false;
    const timer = setTimeout(() => { timedOut = true; child.kill(); }, timeout);
    child.stdout.on('data', chunk => { size += chunk.length; if (size > 96 * 1024 * 1024) child.kill(); else out.push(chunk); });
    child.stderr.on('data', chunk => { size += chunk.length; if (size > 96 * 1024 * 1024) child.kill(); else err.push(chunk); });
    child.once('error', () => { clearTimeout(timer); reject(Error('M07_LOCAL_TRANSPORT_FAILED')); });
    child.once('close', code => { clearTimeout(timer); const stdout = Buffer.concat(out), stderr = Buffer.concat(err).toString('utf8');
      if (timedOut || code !== 0 && !allowFailure) reject(Object.assign(Error('M07_LOCAL_COMMAND_UNCONFIRMED'), { privateDiagnostic: { code, timedOut, stderr } }));
      else resolve({ code, stdout: binary ? stdout : stdout.toString('utf8'), stderr }); });
    child.stdin.on('error', () => {}); child.stdin.end(input);
  });
}
